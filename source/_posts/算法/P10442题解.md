## 题意

简述：

给定字符串 。

## 代码

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    vector<int> count_s(26, 0), count_t(26, 0);

    for (char c : s) {
        count_s[c - 'a']++;
    }

    for (char c : t) {
        count_t[c - 'a']++;
    }

    int ans = 0;
    for (int i = 0; i < 26; i++) {
        ans += min(count_s[i], count_t[i]);
    }
    cout << ans << endl;

    return 0;
}
```
