## 代码

### C++

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll gcd(int x,int y){
    while(x%y){
        int tmp=x%y;
        x=y;
        y=tmp;
    }
    return y;
}
ll lcm(int x,int y){
    return x*y/gcd(x,y);
}

int main(){
    ll x,y;
    cin>>x>>y;
    ll count=0;
     for (int i = x; i <= y; i ++) { 
        int j = x * y / i;      
        if (gcd(i, j) == x && lcm(i, j) == y) {
            count++;
        }
    }
    cout<<count;
    return 0;
}
```
