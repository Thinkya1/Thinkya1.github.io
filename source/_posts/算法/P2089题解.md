## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,kind,m1[10000][10],m2[10];
void func(int total,int a){
    if(a==10){
        if(total==n){
            for (int j = 0; j < 10; j++) {
                m1[kind][j] = m2[j];
            }   
            kind++;
        }
        return ;
    }
    if(total>=n)    return;
    for(int i=1;i<=3;i++){
        m2[a]=i;
        func(total+i,a+1);
    }     
}
int main(){
    cin>>n;
    if (n < 10 || n > 30) {
        cout <<"0"<< endl;
        return 0;
    }
    func(0,0);
    cout<<kind<<endl;
    for(int i=0;i<kind;i++){
        for(int j=0;j<10;j++){
            cout<<m1[i][j]<<" ";
        }
        cout<<endl;
    }
    return 0;
}
```
