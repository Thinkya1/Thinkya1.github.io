## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
int sum[200005],a[200005];
int main(){
    int n,ans=-2147483647;
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        sum[i]=sum[i-1]+a[i];
        sum[i]=max(a[i],sum[i]);
        ans=max(ans,sum[i]);
    }
    cout<<ans;
    return 0;
}
```
