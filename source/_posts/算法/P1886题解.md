#include<bits/stdc++.h>
using namespace std;
int n,k;
deque<int>q;
int a[1000005];
int main(){
    cin>>n>>k;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    for(int i=0;i<n;i++){
        if(!q.empty()&&q.front()<=i-k){
            q.pop_front();
        }
        while(!q.empty()&&a[q.back()]>=a[i]){
            q.pop_back();
        }
        q.push_back(i);
        if(i>=k-1){
            cout<<a[q.front()]<<" ";
        }
    }
    cout<<endl;
    q.clear();
    for(int i=0;i<n;i++){
