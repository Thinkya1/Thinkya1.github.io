## 代码

### C++

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,a[25],k,ans=0,sum[25];
bool check(ll x){
    if(x==1)    return false;
    for(ll i=2;i<x;i++){
        if(x%i==0)  return false;
    }
    return true;
}

void solve(ll total,ll pos,ll start){
    if(pos==k){
        if(check(total)){
            ans++;
        }
        return ;
    }
    
    for(int i=start;i<n;i++){
        solve(total+a[i],pos+1,i+1);
    }
    return ;
}

int main(){
    cin>>n>>k;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    solve(0,0,0);
    cout<<ans;
    return 0;
}
```
