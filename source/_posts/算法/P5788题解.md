#include<bits/stdc++.h>
using namespace std;
int s[3000005],ans[3000005];
int main(){
    int n;
    scanf("%d",&n);
    for(int i=1;i<=n;i++){
        scanf("%d",&s[i]);
    }
    stack<int> st;
    for(int i=n;i>=1;i--){
        while(!st.empty()&&s[st.top()]<=s[i]){      
            st.pop();
        }
        if(st.empty()){
            ans[i]=0;
        }else{
            ans[i]=st.top();
        }
        st.push(i);
    }
    for(int i=1;i<=n;i++){
        printf("%d ",ans[i]);
    }
    return 0;
}
