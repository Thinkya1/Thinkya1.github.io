## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=1e6+10;
ll a[N],s[N];
int main(){
    ll n,k;
    cin>>n>>k;
    if(k+1>=n){
        cout<<0;
        return 0;
    }
    for(int i=1;i<=n-1;i++){
        cin>>a[i];
        s[i]=s[i-1]+a[i];
    }
    ll ans=LLONG_MAX;
    for(int i=1;i<=n-k;i++){
        ll sum = s[i + k - 1] - s[i - 1];
        ans=min(ans,s[n-1]-sum);
    }
    cout<<ans;
    return 0;
}
```
